//------------------------------------------------------------------------------
// MOVEMENT — STARTING MOVES
//------------------------------------------------------------------------------
// MoveTo(const FAIMoveRequest&, FNavPathSharedPtr* OutPath = nullptr)
//     Full control. Build an FAIMoveRequest with goal, radius, partial path, etc.
//
// MoveToActor(AActor* Goal, AcceptanceRadius, bStopOnOverlap, bUsePathfinding,
//             bCanStrafe, FilterClass, bAllowPartialPath)
//     Chase a moving target. Repaths automatically as the target moves.
//     USE FOR: survivor following an ally, fleeing toward a safe actor.
//
// MoveToLocation(FVector Dest, AcceptanceRadius, bStopOnOverlap, bUsePathfinding,
//                bProjectDestinationToNavigation, bCanStrafe, FilterClass,
//                bAllowPartialPath)
//     One-shot move to a fixed point.
//     USE FOR: heading to last known safe spot, scavenging waypoint.
//
// Returns: EPathFollowingRequestResult::Type
//     - RequestSuccessful   (path computed, moving)
//     - AlreadyAtGoal       (no movement needed)
//     - Failed              (no navmesh, no path, invalid goal)


//------------------------------------------------------------------------------
// MOVEMENT — STOPPING / PAUSING
//------------------------------------------------------------------------------
// StopMovement()                  Hard stop, clears current path.
// PauseMove(FAIRequestID)         Pause without clearing (e.g. wait for door).
// ResumeMove(FAIRequestID)        Resume a paused move.

// REACTING TO MOVE COMPLETION
//------------------------------------------------------------------------------
// OPTION 1 — Override in your AAIController subclass:
//
// virtual void OnMoveCompleted(FAIRequestID RequestID,
//                              const FPathFollowingResult& Result) override
// {
//     Super::OnMoveCompleted(RequestID, Result);
//
//     if (Result.IsSuccess())              { /* reached goal */ }
//     else if (Result.IsInterrupted())     { /* replaced by another move */ }
//     else if (Result.HasFlag(FPathFollowingResultFlags::UserAbort)) { /* StopMovement */ }
//     else if (Result.HasFlag(FPathFollowingResultFlags::Blocked))   { /* zombie blocking? */ }
//     else if (Result.HasFlag(FPathFollowingResultFlags::InvalidPath)) { /* no path */ }
// }
//
// OPTION 2 — Bind from external component:
//
// if (UPathFollowingComponent* PFComp = m_AIController->GetPathFollowingComponent())
// {
//     PFComp->OnRequestFinished.AddUObject(this, &UMyComp::HandleMoveFinished);
// }
//
// REMEMBER to unbind in EndPlay if listener can outlive the controller.
